// Form Submit Logic (Menggunakan GET Request untuk menghindari kendala CORS) document.getElementById('bkSurveyForm').addEventListener('submit', function(e) { e.preventDefault(); const tingkat = document.getElementById('tingkatSelect').value; const answers = []; for (let i = 1; i <= 12; i++) { const selected = document.querySelector(`input[name="p_${i}"]:checked`); answers.push(selected ? selected.value : '-'); } const payload = { timestamp: new Date().toLocaleString("id-ID"), nama: document.getElementById('nama').value, tingkat: tingkat, rombel: document.getElementById('rombelSelect').value, jk: document.getElementById('jk').value, p1: answers[0], p2: answers[1], p3: answers[2], p4: answers[3], p5: answers[4], p6: answers[5], p7: answers[6], p8: answers[7], p9: answers[8], p10: answers[9], p11: answers[10], p12: answers[11], extra: document.getElementById('extraAnswer').value }; const btnSubmit = document.getElementById('btnSubmit'); btnSubmit.disabled = true; btnSubmit.innerHTML = 'Mengirim... '; // Buat URL Query Parameters const params = new URLSearchParams(payload).toString(); // Kirim menggunakan GET Request fetch(`${scriptURL}?${params}`, { method: 'GET' }) .then(res => res.json()) .then(data => { if (data.result === 'success') { localStorage.setItem('bk_survey_submitted', 'true'); alert('Terima kasih! Survey berhasil terkirim ke Database BK SMAN 2 Satui.'); window.location.reload(); } else { // TAMPILKAN PESAN ERROR DARI SERVER JIKA NAMA DUPLIKAT alert(data.message); btnSubmit.disabled = false; btnSubmit.className = "w-full bg-indigo-600 hover:bg-indigo-700 text-white font-semibold py-3.5 px-4 rounded-xl shadow-lg shadow-indigo-600/30 transition flex items-center justify-center space-x-2 cursor-pointer"; btnSubmit.innerHTML = 'Kirim Jawaban Survey '; } }) .catch(err => { alert('Terjadi kesalahan koneksi. Pastikan izin Apps Script diset ke "Anyone" (Siapa Saja).'); btnSubmit.disabled = false; btnSubmit.className = "w-full bg-indigo-600 hover:bg-indigo-700 text-white font-semibold py-3.5 px-4 rounded-xl shadow-lg shadow-indigo-600/30 transition flex items-center justify-center space-x-2 cursor-pointer"; btnSubmit.innerHTML = 'Kirim Jawaban Survey '; }); });
